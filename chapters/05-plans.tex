\chapter{إعداد خطة التكليف وإطلاقها}
\label{ch:05-plans}
\section{ما خطة التكليف؟}
\label{sec:plans-what}

الحزمة تحدد ماذا يُعبّأ. أما الخطة فتحدد لأي منشآت، ومن يعبّئ، ومتى ينتهي العمل. وعند إطلاق الخطة يُنشأ تكليف واحد لكل منشأة، وفي داخله وحدات الحزمة كلها. لذلك تغنيك الخطة عن تكليف المنشآت واحدة واحدة، وتجمع متابعتها في لوحة واحدة.

\textbf{أنشئ الخطة من الحزمة:}

\begin{enumerate}
\item افتح الحزمة المنشورة، واضغط «التكليف» في شريطها العلوي.
\item اضغط «خطة جديدة». تُفتح الخطة بحالة «مسودة» على تبويب «الإعداد».
\end{enumerate}

لا تخلط هذه الخطة بالخطط العامة في تطبيق المشاريع. وتجد خططك لاحقاً في «المهام» ثم «التكليفات» ثم «الخطط».

\centering
\includegraphics[width=\linewidth]{assets/screens/plans-setup-ar.jpg}
\caption{صفحة إعداد خطة جديدة}
\label{fig:plans-setup}
\shotnote{هذه الصفحة التي تبني فيها الخطة كلها قبل إطلاقها. يُطوى كل قسم على سطر يلخص اختيارك، فتراجع الإعداد كله دون أن تفتح الأقسام. ولا يصل شيء إلى الباحثين قبل «إنشاء التكليفات». ① اسم الخطة وحالتها «مسودة». ② ملخص الخطة: جملة واحدة تجمع النطاق والتوزيع والموعد، وتتحدث مع كل تعديل. ③ «الإعداد». ④ الأقسام الستة و«متقدم». ⑤ «معاينة قبل الإطلاق»: تحسب النتيجة دون أن تنشئ شيئاً. ⑥ «إنشاء التكليفات»: الإطلاق الفعلي، وبجانبه «حفظ».}
\end{figure}

نشرح في الأقسام التالية كل حقل: ماذا يفعل، وما الذي تكسبه منه. وفي آخر كل قسم نطبّقه على خطة مصنع الأفق.

\begin{rcnote}[تطبيق المثال]
تفتح سارة حزمة «مسح تشغيل الأفق» التي نشرتها في الفصل الرابع، ثم تضغط «التكليف» ثم «خطة جديدة». تظهر الخطة «مسودة»، والأقسام كلها فارغة.
\end{rcnote}

\section{«1 · السجلات»: أي منشآت ومتى؟}
\label{sec:plans-records}

يحدد هذا القسم نطاق المسح: ماذا يُعبّأ، وعن أي منشآت، ومتى تُنشأ التكليفات.

\centering
\includegraphics[width=\linewidth]{assets/screens/plans-s1-ar.png}

\end{figure}

\thh{الحقل} & \thh{ماذا يفعل؟} & \thh{الفائدة}\\
«ما يُكلَّف به» & يعرض الحزمة التي فتحت منها الخطة، وعدد وحداتها وإصدارها. & تتأكد أنك تخطط للحزمة الصحيحة.\\
«نوع السجل» & يعرض نوع السجل الذي يشير إليه سؤال المنشأة في النماذج. & يضمن أن كل تكليف يرتبط بملف منشأة.\\

«السجلات» & «كل السجلات المطابقة» تأخذ كل منشأة تحقق الشروط. «اختيار سجلات» تأخذ المنشآت التي تحددها بأسمائها. & الشروط تناسب المسح الواسع، والاختيار يناسب التجربة أو عدداً صغيراً.\\
«أي السجلات تُكلَّف» & شروط على حقول المنشأة، بطريقة «بصري» أو «كود». مثال: المحافظة = ريف دمشق، وحالة التشغيل = عاملة. & تستبعد المنشآت المغلقة أو الخارجة عن نطاق المسح قبل أن تُكلَّف.\\
\end{tabularx}
\end{table}

راقب «معاينة قبل الإطلاق» في الجانب أثناء العمل. فهي تعرض عدد المنشآت المطابقة وأسماءها، وتتحدث مع كل تعديل في الشروط.

\begin{rcnote}[تطبيق المثال]
تختار سارة «عند الإطلاق»؛ فجولة أيلول تُطلق مرة واحدة. ثم تختار «اختيار سجلات»، وتحدد «مصنع الأفق للصناعات الغذائية» وحده. تعرض المعاينة «1» تطابق، وتحتها اسم المصنع. ولو كان المسح لكل مصانع ريف دمشق، لاختارت «كل السجلات المطابقة» وأضافت شرط المحافظة.
\end{rcnote}

\section{«2 · المرشحون»: من أين يأتي الأشخاص؟}
\label{sec:plans-candidates}

المرشحون هم الأشخاص الذين يجوز تكليفهم. تجمعهم الخطة لكل منشأة على حدة، ثم يختار قسم التوزيع واحداً منهم.

\centering
\includegraphics[width=\linewidth]{assets/screens/plans-s2-ar.png}
\caption{قسم «2 · المرشحون»}
\label{fig:plans-candidates}
\shotnote{يحدد هذا القسم من يمكن أن يصله التكليف. في الصورة اختيرت «قائمة ثابتة» بعضو واحد أُخفي اسمه، أما مثالنا فيستخدم «حقل في السجل» كما في الفصل الثاني. ① «المرشحون من». ② العضو المختار. ③ «إن لم يُوجد أحد»: هنا «يبقى دون إسناد».}
\end{figure}

\centering
\caption{حقول «2 · المرشحون»}

\begin{tabularx}{\linewidth}{R{3.2cm} Y Y}
\rcHeaderRow
\thh{الحقل} & \thh{ماذا يفعل؟} & \thh{الفائدة}\\
«المرشحون من»: «حقل في السجل» & يقرأ حقلاً في ملف المنشأة، مثل «فريق المسح»، ثم يأخذ «شاغلو هذه المناصب فيها»، مثل «باحث ميداني - RES». & يتبع التكليف الفريق المسؤول عن المنشأة. فإذا نُقلت المنشأة إلى فريق آخر، تغيّر مرشحوها دون تعديل الخطة.\\
«قائمة ثابتة» & تختار «أعضاء، مناصب، وحدات» بأسمائها. & الأسرع لتجربة أو فريق صغير ثابت.\\
«سجلات تشير إليه» & يأخذ الأشخاص من سجلات أخرى ترتبط بالمنشأة، مثل جدول زيارات المفتشين. & يتبع التكليف جدولاً قائماً؛ فإضافة شخص فيه تحدّث التكليفات المفتوحة.\\
«بالكود» & تعبير يُحسب لكل منشأة ويُرجع الأشخاص أو المناصب. & لقواعد خاصة لا تغطيها الطرق الأخرى.\\
«إن لم يُوجد أحد» & ما يحدث لمنشأة بلا مرشح: «يبقى دون إسناد»، أو «الوحدة الأعلى»، أو «قائمة الفريق»، أو «شخص محدد». & لا تضيع منشأة بصمت؛ فإما أن تظهر لك في «دون إسناد»، أو تذهب إلى بديل تحدده.\\
\end{tabularx}
\end{table}

\begin{rcnote}[تطبيق المثال]
تختار سارة «حقل في السجل»، ثم الحقل «فريق المسح»، ثم منصب «باحث ميداني - RES». وفي «إن لم يُوجد أحد» تختار «يبقى دون إسناد»، لتعالج أي منشأة بلا باحث بنفسها. عمر هو الباحث الوحيد في فريق مسح الأفق، فيصبح مرشح المصنع الوحيد.
\end{rcnote}

«تخطّي الغائبين» & يتجاوز من لمنصبه تفويض نافذ. & لا يصل التكليف إلى شخص في إجازة.\\

\end{table}

\end{rcnote}

\section{«4 · الموعد والتذكير»: متى ينتهي العمل؟}
\label{sec:plans-due}

\end{tabularx}
\end{table}

\begin{rcnote}[تطبيق المثال]

\end{rcnote}

«الأعداد المتوقعة» & «بلا»، أو «لكل مجموعة» تكتبه بنفسك، أو «من حقل في السجل» يُقرأ من سجل مرتبط. & تحدد هدفاً واضحاً لكل مجموعة، لا عدداً عاماً فقط.\\
«العدد المتوقع» & الهدف لكل مجموعة. اتركه فارغاً لمجموعة بلا هدف، ويُنسخ إلى الجولة التالية. & تعرف النسبة المنجزة من الهدف في كل محافظة.\\
«يُحتسب نحو العدد عندما تصبح الإجابة» & «مُرسلة»، أو «مقبولة»، أو «كما في «ينتهي التكليف»». & لا تحتسب بيانات لم تُراجع ضمن الإنجاز.\\
«عند بلوغ العدد» & «إيقاف إنشاء تكليفات جديدة لهذه المجموعة»: تبقى التكليفات المفتوحة، ولا تُنشأ تكليفات جديدة للمجموعة حتى الجولة التالية. & يفيد في مسح العينات: تتوقف المجموعة عند اكتمال عينتها.\\
\end{tabularx}
\end{table}

\begin{rcnote}[تطبيق المثال]
تختار سارة «تجميع حسب» «المحافظة»، ثم «لكل مجموعة»، وتحفظ الخطة ليظهر جدول المحافظات. تكتب العدد المتوقع لريف دمشق 1، وتختار أن يُحتسب عند «مقبولة».
\end{rcnote}

\section{«6 · من يدير الخطة» و«متقدم»}
\label{sec:plans-advanced}

يحدد القسم السادس من يملك الخطة. أما «متقدم» فيضبط فترة الجولة وإصدار النماذج.

\begin{figure}[H]
\centering
\includegraphics[width=\linewidth]{assets/screens/plans-s6-ar.png}
\caption{قسما «6 · من يدير الخطة» و«متقدم»}

\shotnote{يجمع الجزء الأعلى أسماء المسؤولين عن الخطة، ويضبط الجزء الأدنى ما يُحفظ مع كل جولة. ويظهر تحت «فترة الجولة» ما يستطيع النموذج قراءته من الجولة. ① «الاسم». ② «مسؤولو الخطة»، وقد أُخفي الاسم. ③ «من يكلّف يدوياً ضمن الخطة». ④ «فترة الجولة»: «تُحدد عند الإطلاق». ⑤ «إصدار النموذج»: «تثبيت عند كل جولة».}
\end{figure}

\begin{table}[H]
\centering
\caption{حقول «6 · من يدير الخطة» و«متقدم»}
\label{tab:plans-advanced-fields}
\begin{tabularx}{\linewidth}{R{3.2cm} Y Y}
\rcHeaderRow
\thh{الحقل} & \thh{ماذا يفعل؟} & \thh{الفائدة}\\
«الاسم» & اسم الخطة في القوائم والإشعارات واللوحات. & اختر اسماً يميّز المسح والفترة.\\
«مسؤولو الخطة» & يعدّلون الخطة، ويعيدون الإسناد، ويرون لوحتها. & تتوزع إدارة المسح دون منح صلاحيات أوسع.\\
«من يكلّف يدوياً ضمن الخطة» & أشخاص يكلّفون منشأة بأنفسهم، من قائمة السجلات أو من لوحة الفريق. & يعالج قائد الفريق حالة خاصة دون أن يعدّل الخطة.\\
«فترة الجولة» & «رقم الجولة فقط»، أو «من القاعدة» للمسح الدوري، أو «تُحدد عند الإطلاق» فيُطلب الاسم والتاريخان. & تقرأ أسئلة النموذج اسم الجولة وتاريخيها، مثل «فترة المسح».\\

ثم اضغط «حفظ». يظهر ملخص الخطة في أعلاها بجملة واحدة، فاقرأه للتأكد من الإعداد كله.

\begin{rcnote}[تطبيق المثال]
تكتب سارة الاسم «متابعة تشغيل الأفق»، وتضيف نفسها في «مسؤولو الخطة». وفي «متقدم» تختار «تُحدد عند الإطلاق» و«تثبيت عند كل جولة»، ثم تحفظ. يظهر الملخص: «1 من المنشآت الصناعية محددة ← شخص واحد … الأقل حملاً · بحد 15 لكل شخص · الموعد 14 أيام عمل».
\end{rcnote}

\section{عاين قبل الإطلاق}
\label{sec:plans-preview}

اضغط «تشغيل المعاينة». تعرض المعاينة ثلاثة أعداد: «سجلات مطابقة»، و«تكليفات» ستُنشأ، و«دون مرشح». وتعرض أيضاً حِمل كل مرشح، وسبب بقاء أي منشأة دون إسناد، ومثالاً على ثلاث منشآت. المعاينة لا تنشئ شيئاً.

\section{أطلق الجولة وتحقق منها}
\label{sec:plans-launch}

\begin{enumerate}
\item اضغط «إنشاء التكليفات».
\item يطلب الإطلاق «اسم الجولة» وتاريخي «من» و«إلى»، لأن «فترة الجولة» = «تُحدد عند الإطلاق».
\item اضغط «إنشاء التكليفات» في النافذة. تصل الإشعارات إلى المكلَّفين فوراً.
\end{enumerate}

\begin{figure}[H]
\centering
\includegraphics[width=0.62\linewidth]{assets/screens/plans-launch-ar.png}
\caption{نافذة إطلاق الجولة}
\label{fig:plans-launch}
\shotnote{هذه آخر خطوة قبل أن يصل العمل إلى الباحثين. تفيدك في تسمية الجولة وتحديد فترتها، وتقرأ أسئلة النموذج هذه القيم. ① «اسم الجولة». ② «من». ③ «إلى». ④ «إنشاء التكليفات».}
\end{figure}

\textbf{ما الذي يُنشأ عند الإطلاق؟} تصبح الخطة «نشطة»، وتبدأ «الجولة 1». ويُنشأ لكل منشأة تكليف واحد، و«سجل الجولة» الخاص بها. وتُفتح الوحدة الأولى فقط؛ أما بقية الوحدات فتنتظر شروط فتحها.

\begin{figure}[H]
\centering
\includegraphics[width=\linewidth]{assets/screens/plans-overview-ar.jpg}
\caption{«نظرة عامة» بعد إطلاق خطة المثال}
\label{fig:plans-overview}
\shotnote{هذه لوحة الخطة بعد الإطلاق مباشرة. تفيدك في التحقق من أن كل منشأة مستهدفة صار لها تكليف، وأن الوحدة الأولى مفتوحة. ① الحالة «نشطة» واسم الجولة. ② الأعداد: «السجلات المستهدفة» و«مُسندة» و«بدأت» و«مكتملة» و«معتمدة» و«متأخرة». ③ «الجولات» وفترتها. ④ «الوحدات»: الوحدة الأولى «مفتوحة». ⑤ «الإنجاز حسب الشخص»، وقد أُخفي الاسم.}
\end{figure}

\textbf{إذا عدّلت نموذجاً بعد الإطلاق:} تبقى الجولة على إصدار الحزمة الذي بدأت به. انشر النموذج، ثم انشر الحزمة، ثم اضغط «نقل الجولة إلى الأحدث» في «متقدم». والإجابات التي بدأت تبقى على إصدارها في كل الأحوال.

\begin{rcnote}[تطبيق المثال]
تكتب سارة «أيلول 2026»، والفترة من 1 إلى 30 أيلول 2026، ثم تطلق. تصبح الخطة «نشطة · الجولة 1 · أيلول 2026»، وتعرض «السجلات المستهدفة 1» و«مُسندة 1»، والوحدة «بيانات التشغيل» «مفتوحة». ويُنشأ سجل جولة «أيلول 2026» لمصنع الأفق.
\end{rcnote}

\begin{rcchecklist}[قبل أن يبدأ عمر]
\begin{checklist}
\item الخطة «متابعة تشغيل الأفق» مرتبطة بحزمة «مسح تشغيل الأفق».
\item النطاق مصنع الأفق وحده، والمرشح من «فريق المسح».
\item التوزيع: شخص واحد، الأقل حِملاً، حد 15.
\item الموعد 14 يوم عمل، وينتهي التكليف عندما تصبح الإجابة «مقبولة».
\item المتابعة حسب المحافظة، والعدد المتوقع لريف دمشق 1.
\item المعاينة: 1 تكليف و0 دون مرشح.
\item الخطة «نشطة»، والجولة «أيلول 2026»، والوحدة الأولى «مفتوحة».
\end{checklist}
\end{rcchecklist}

وصل التكليف الآن إلى عمر. في الفصل~\ref{ch:06-assignments} يفتحه ويعبّئ الوحدتين.
